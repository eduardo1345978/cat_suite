# Stress test: each plug-in's shared code + the real editor, random UI + audio at once.
# Build with -DCAT_BUILD_STRESS=ON -DCMAKE_CXX_FLAGS="-fsanitize=address -D_GLIBCXX_ASSERTIONS" and run under Xvfb.
option(CAT_BUILD_STRESS "Build the per-plug-in stress test executables" OFF)
if(CAT_BUILD_STRESS)
    set(_stress ChordCat PattCat BendCat DrumCat OrbitCat SingCat LifeCat BassCat GlideCat TalkCat MuseCat EchoCat MicroCat PhaseCat HumCat GlitchCat Molten Swarm Negative Vortex Crypt CryptFX PolyBoard)
    if(CAT_BUILD_PRO)
        list(APPEND _stress ChordCatPro GlideCatPro VortexPro NegativePro CryptPro)    # the Pro panel, with few and many parameters
    endif()
    foreach(_cat IN LISTS _stress)
        add_executable(${_cat}Stress Tests/PluginStress.cpp)
        target_compile_definitions(${_cat}Stress PRIVATE $<TARGET_PROPERTY:${_cat},COMPILE_DEFINITIONS>)
        target_include_directories(${_cat}Stress PRIVATE $<TARGET_PROPERTY:${_cat},INCLUDE_DIRECTORIES>)
        target_compile_features(${_cat}Stress PRIVATE cxx_std_20)
        target_link_libraries(${_cat}Stress PRIVATE ${_cat})
    endforeach()
    add_executable(MicroCheck Tests/MicroCheck.cpp ThirdParty/MTS-ESP/Client/libMTSClient.cpp)
    set_source_files_properties(ThirdParty/MTS-ESP/Client/libMTSClient.cpp PROPERTIES COMPILE_OPTIONS "-w")
    target_compile_definitions(MicroCheck PRIVATE $<TARGET_PROPERTY:MicroCat,COMPILE_DEFINITIONS>)
    target_include_directories(MicroCheck PRIVATE $<TARGET_PROPERTY:MicroCat,INCLUDE_DIRECTORIES>)
    target_compile_features(MicroCheck PRIVATE cxx_std_20)
    target_link_libraries(MicroCheck PRIVATE MicroCat)
    add_executable(MoltenCheck Tests/MoltenCheck.cpp)
    target_compile_definitions(MoltenCheck PRIVATE $<TARGET_PROPERTY:Molten,COMPILE_DEFINITIONS>)
    target_include_directories(MoltenCheck PRIVATE $<TARGET_PROPERTY:Molten,INCLUDE_DIRECTORIES>)
    target_compile_features(MoltenCheck PRIVATE cxx_std_20)
    target_link_libraries(MoltenCheck PRIVATE Molten)
    add_executable(SwarmCheck Tests/SwarmCheck.cpp)
    target_compile_definitions(SwarmCheck PRIVATE $<TARGET_PROPERTY:Swarm,COMPILE_DEFINITIONS>)
    target_include_directories(SwarmCheck PRIVATE $<TARGET_PROPERTY:Swarm,INCLUDE_DIRECTORIES>)
    target_compile_features(SwarmCheck PRIVATE cxx_std_20)
    target_link_libraries(SwarmCheck PRIVATE Swarm)
    add_executable(NegativeCheck Tests/NegativeCheck.cpp)
    target_compile_definitions(NegativeCheck PRIVATE $<TARGET_PROPERTY:Negative,COMPILE_DEFINITIONS>)
    target_include_directories(NegativeCheck PRIVATE $<TARGET_PROPERTY:Negative,INCLUDE_DIRECTORIES>)
    target_compile_features(NegativeCheck PRIVATE cxx_std_20)

    add_executable(LoopCheck Tests/LoopCheck.cpp apps/CatLoop/Source/LoopEngine.cpp Shared/Source/CatSounds.cpp midi/DrumCat/Source/DrumCatEngine.cpp)
    target_include_directories(LoopCheck PRIVATE apps/CatLoop/Source Shared/Source midi/DrumCat/Source)
    target_compile_features(LoopCheck PRIVATE cxx_std_20)
    target_compile_definitions(LoopCheck PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(LoopCheck PRIVATE juce::juce_audio_basics juce::juce_data_structures juce::juce_events juce::juce_gui_basics juce::juce_recommended_config_flags)
    add_executable(VortexCheck Tests/VortexCheck.cpp)
    target_compile_definitions(VortexCheck PRIVATE $<TARGET_PROPERTY:Vortex,COMPILE_DEFINITIONS>)
